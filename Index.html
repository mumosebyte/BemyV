<div class="container">
    <h1 id="question">Will you be my Valentine, Gijon? 💘</h1>

    <select id="nameSelect" onchange="updateName()">
        <option value="Gijon">Gijon</option>
        <option value="Digga">Digga</option>
        <option value="Bophelo">Bophelo</option>
    </select>

    <div class="buttons">
        <button id="yesBtn" onclick="sayYes()">Yes 💖</button>
        <button id="noBtn"
            onclick="escapeNo()"
            ontouchstart="escapeNo()">No 🙃</button>
    </div>

    <div id="message"></div>
</div>

<script>
    function updateName() {
        const name = document.getElementById("nameSelect").value;
        document.getElementById("question").innerText =
            `Will you be my Valentine, ${name}? 💘`;
    }

    function escapeNo() {
        const noBtn = document.getElementById("noBtn");

        const padding = 20;
        const btnWidth = noBtn.offsetWidth;
        const btnHeight = noBtn.offsetHeight;

        const maxX = window.innerWidth - btnWidth - padding;
        const maxY = window.innerHeight - btnHeight - padding;

        const x = Math.random() * maxX;
        const y = Math.random() * maxY;

        noBtn.style.left = x + "px";
        noBtn.style.top = y + "px";

        alert("😏 Are you sure? Try pressing YES!");
    }

    function sayYes() {
        document.getElementById("message").style.display = "block";
        document.getElementById("message").innerHTML =
            "💖 I knew you would say YES!<br>I also knew you wouldn’t press the NO button 😄";
    }

    // Floating hearts generator
    for (let i = 0; i < 14; i++) {
        const heart = document.createElement("div");
        heart.className = "heart";
        heart.style.left = Math.random() * 100 + "vw";
        heart.style.animationDuration = (6 + Math.random() * 6) + "s";
        heart.style.opacity = Math.random();
        document.body.appendChild(heart);
    }
</script>
